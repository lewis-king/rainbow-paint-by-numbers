import cv2
import numpy as np
import os
import json
import shutil
import argparse
import re
from sklearn.cluster import KMeans
from processing_resources import desktop_friendly

# CONFIG
INPUT_DIR = "raw_assets"
OUTPUT_DIR = "app_assets"
TARGET_SIZE = (1024, 1024) 

INITIAL_K = 24       
MERGE_THRESHOLD = 60  # IMPROVED: was 40 - better color merging
BACKGROUND_ID = 255  
LINE_ID = 254        

# === IMPROVED THRESHOLDS ===
OUTLINE_THICKNESS_THRESHOLD = 30
NEIGHBOR_GAP = 20     # IMPROVED: was 15 - better region grouping  
MIN_REGION_AREA = 20
MIN_REGION_WIDTH = 8
DESPECKLE_SIZE = 5    # IMPROVED: was 3 - smoother regions
BEGINNER_MODE = False

def write_rgba_png(path, rgba):
    # OpenCV's encoder consumes BGRA; fixed colored details must retain RGB hues.
    cv2.imwrite(path, cv2.cvtColor(rgba, cv2.COLOR_RGBA2BGRA))

def simplify_beginner_map(map_img, centers, min_radius=20):
    """Keep number targets large enough to read; absorb thin raster edge fragments.

    Small details use LINE_ID and are rendered as fixed artwork by the caller.
    Large black areas remain paintable.
    Compact the palette so discarded anti-alias shades never become game choices.
    This is opt-in and does not alter previously shipped levels.
    """
    result = map_img.copy()
    for uid in np.unique(map_img):
        if uid >= LINE_ID:
            continue
        count, labels, stats, _ = cv2.connectedComponentsWithStats(
            np.uint8(map_img == uid), connectivity=4)
        for component in range(1, count):
            x, y, w, h, _ = stats[component]
            mask = np.uint8(labels[y:y+h, x:x+w] == component)
            padded = np.pad(mask, 1)
            radius = cv2.distanceTransform(padded, cv2.DIST_L2, 5).max()
            if radius < min_radius:
                crop = result[y:y+h, x:x+w]
                crop[mask > 0] = LINE_ID
    used = [int(uid) for uid in np.unique(result) if uid < LINE_ID]
    compact = result.copy()
    for new, old in enumerate(used):
        compact[result == old] = new
    return compact, [centers[old] for old in used]

def rgb_to_hex(rgb):
    return '#{:02x}{:02x}{:02x}'.format(int(rgb[0]), int(rgb[1]), int(rgb[2]))

def color_distance(c1, c2):
    return np.sqrt(np.sum((c1 - c2) ** 2))

def merge_palette_centers(centers, beginner=False):
    """Merge shades without turning light gray into pink or white.

    Beginner assets compare perceptual Lab colours. Legacy mode keeps its
    historical RGB merge for the existing assets.
    """
    centers = np.asarray(centers, dtype=np.float64)
    distances = (cv2.cvtColor(centers.astype(np.float32).reshape(1, -1, 3) / 255,
                             cv2.COLOR_RGB2LAB)[0] if beginner else centers)
    threshold = 12 if beginner else MERGE_THRESHOLD
    order = np.argsort(np.sum(centers, axis=1))
    palette, mapping = [], {}
    for i in order:
        if i in mapping:
            continue
        index = len(palette)
        palette.append(centers[i])
        mapping[i] = index
        for j in order:
            if j not in mapping and color_distance(distances[i], distances[j]) < threshold:
                mapping[j] = index
    return palette, mapping


def enclosed_background(map_img):
    """Find background pockets surrounded by ink, excluding the outer canvas."""
    _, components = cv2.connectedComponents(np.uint8(map_img == BACKGROUND_ID), connectivity=4)
    outside = np.unique(np.concatenate((components[0], components[-1],
                                        components[:, 0], components[:, -1], [0])))
    return ~np.isin(components, outside)

def get_luminance(rgb):
    return 0.299*rgb[0] + 0.587*rgb[1] + 0.114*rgb[2]

def get_pole_of_inaccessibility(mask):
    dist_transform = cv2.distanceTransform(mask, cv2.DIST_L2, 5)
    _, max_val, _, max_loc = cv2.minMaxLoc(dist_transform)
    return max_loc[0], max_loc[1], max_val

def check_region_width_height(mask):
    ys, xs = np.where(mask > 0)
    if len(ys) == 0 or len(xs) == 0:
        return False, 0, 0
    width = np.max(xs) - np.min(xs)
    height = np.max(ys) - np.min(ys)
    return width >= MIN_REGION_WIDTH and height >= MIN_REGION_WIDTH, width, height 

def group_neighboring_regions(map_img, color_idx, min_area, neighbor_gap):
    mask = np.uint8(map_img == color_idx) * 255
    num_labels, labels, stats, _ = cv2.connectedComponentsWithStats(mask, connectivity=4 if BEGINNER_MODE else 8)
    original_regions = []
    
    for i in range(1, num_labels):
        if stats[i, cv2.CC_STAT_AREA] >= min_area:
            region_mask = np.uint8(labels == i) * 255
            is_valid, w, h = check_region_width_height(region_mask)
            if is_valid:
                original_regions.append(i) 

    if not original_regions:
        return []

    if neighbor_gap == 0:
        result = []
        for component in original_regions:
            component_mask = np.uint8(labels == component) * 255
            x, y, _ = get_pole_of_inaccessibility(component_mask)
            area = int(stats[component, cv2.CC_STAT_AREA])
            result.append({'regions': [{'mask': component_mask, 'area': area}],
                           'best_label_pos': (x, y), 'total_area': area})
        return result

    kernel_size = max(3, neighbor_gap)
    kernel = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (kernel_size, kernel_size))
    bridged_mask = cv2.dilate(mask, kernel)
    
    num_groups, bridged_labels, _, _ = cv2.connectedComponentsWithStats(bridged_mask, connectivity=8)
    
    groups = {} 
    for orig_label_id in original_regions:
        ys, xs = np.where(labels == orig_label_id)
        if len(ys) == 0: continue
        
        group_id = bridged_labels[ys[0], xs[0]]
        if group_id == 0: continue

        if group_id not in groups:
            groups[group_id] = []
        
        component_mask = np.uint8(labels == orig_label_id) * 255
        area = stats[orig_label_id, cv2.CC_STAT_AREA]
        
        groups[group_id].append({
            'mask': component_mask,
            'area': area
        })
        
    result_groups = []
    for gid, region_list in groups.items():
        total_area = sum(r['area'] for r in region_list)
        largest_region = max(region_list, key=lambda x: x['area'])
        cx, cy, _ = get_pole_of_inaccessibility(largest_region['mask'])
        
        result_groups.append({
            'regions': region_list,
            'best_label_pos': (cx, cy),
            'total_area': total_area
        })
        
    return result_groups

def remove_small_regions(map_img, min_area):
    h, w = map_img.shape
    unique_ids = np.unique(map_img)
    total_cleaned = 0
    
    for uid in unique_ids:
        if uid == BACKGROUND_ID: continue
        mask = np.uint8(map_img == uid) * 255
        num_labels, labels, stats, _ = cv2.connectedComponentsWithStats(mask, connectivity=8)
        
        for i in range(1, num_labels):
            area = stats[i, cv2.CC_STAT_AREA]
            region_mask = np.uint8(labels == i) * 255
            
            is_valid, w, h = check_region_width_height(region_mask)
            if area < min_area or not is_valid:
                blob_mask = np.uint8(labels == i) * 255
                dilated = cv2.dilate(blob_mask, np.ones((3,3), np.uint8))
                neighbor_mask = cv2.bitwise_xor(dilated, blob_mask)
                neighbors = map_img[neighbor_mask > 0]
                
                if len(neighbors) > 0:
                    counts = np.bincount(neighbors)
                    dominant_neighbor = np.argmax(counts)
                    map_img[blob_mask > 0] = dominant_neighbor
                    total_cleaned += 1
    return map_img

@desktop_friendly
def process_level(filename):
    print(f"Processing {filename}...")
    base_name = os.path.splitext(filename)[0]
    
    img_path = os.path.join(INPUT_DIR, filename)
    vid_path = os.path.join(INPUT_DIR, base_name + ".mp4") 
    save_path = os.path.join(OUTPUT_DIR, base_name)
    
    if not os.path.exists(save_path): os.makedirs(save_path)

    img = cv2.imread(img_path)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

    print(f"   - Resizing from {img.shape[:2]} to {TARGET_SIZE}...")
    img = cv2.resize(img, TARGET_SIZE, interpolation=cv2.INTER_AREA)
    source_ink = None
    if BEGINNER_MODE:
        # Mean-shift filtering can erase a thin smile or an internal contour.
        # Recover genuine dark strokes directly from the source before filtering.
        dark = np.uint8(np.max(img, axis=2) < 65) * 255
        broad_dark = cv2.morphologyEx(dark, cv2.MORPH_OPEN,
            cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (31, 31)))
        source_ink = (dark > 0) & (broad_dark == 0)

    print("   - Ironing gradients...")
    img = cv2.pyrMeanShiftFiltering(img, sp=25, sr=60) 
    img = cv2.bilateralFilter(img, 9, 75, 75) 

    pixels = img.reshape((-1, 3))
    kmeans = KMeans(n_clusters=INITIAL_K, random_state=42, n_init=10)
    kmeans.fit(pixels)
    centers = kmeans.cluster_centers_
    labels = kmeans.labels_
    
    print(f"   - K-Means: {INITIAL_K} colors.")

    final_centers, final_map_map = merge_palette_centers(centers, beginner=BEGINNER_MODE)

    print(f"   - Merged to {len(final_centers)} colors.")

    new_labels = np.array([final_map_map[l] for l in labels])
    map_img = new_labels.reshape(img.shape[:2]).astype(np.uint8)
    
    print("   - Melting gaps...")
    kernel = np.ones((3,3), np.uint8) 
    map_img = cv2.morphologyEx(map_img, cv2.MORPH_CLOSE, kernel)
    map_img = cv2.medianBlur(map_img, DESPECKLE_SIZE) 

    min_lum = 255
    darkest_idx = -1
    for idx, color in enumerate(final_centers):
        lum = get_luminance(color)
        if lum < min_lum:
            min_lum = lum
            darkest_idx = idx
            
    has_black_outline = False
    if min_lum < 60: 
        print(f"   - Analyzing Darkest Color (ID: {darkest_idx}, Lum: {min_lum:.1f})...")
        dark_mask = np.uint8(map_img == darkest_idx) * 255
        
        kernel = cv2.getStructuringElement(cv2.MORPH_RECT, (OUTLINE_THICKNESS_THRESHOLD, OUTLINE_THICKNESS_THRESHOLD))
        paintable_dark_parts = cv2.morphologyEx(dark_mask, cv2.MORPH_OPEN, kernel)
        detected_lines = dark_mask - paintable_dark_parts
        
        num_l, labels_l, stats_l, _ = cv2.connectedComponentsWithStats(detected_lines, connectivity=8)
        cleaned_lines = np.zeros_like(detected_lines)
        for i in range(1, num_l):
            if stats_l[i, cv2.CC_STAT_AREA] > 10:
                cleaned_lines[labels_l == i] = 255
        
        if np.count_nonzero(cleaned_lines) > 0:
            map_img[cleaned_lines > 0] = LINE_ID
            has_black_outline = True
            print(f"   - SUCCESS: Detected lines (thinner than {OUTLINE_THICKNESS_THRESHOLD}px).")
        else:
            print("   - WARNING: No lines detected.")

    map_img = remove_small_regions(map_img, MIN_REGION_AREA)
    
    h, w = map_img.shape[:2]
    mask = np.zeros((h+2, w+2), np.uint8)
    cv2.floodFill(map_img, mask, (0,0), BACKGROUND_ID, flags=4 | (255 << 8))
    cv2.floodFill(map_img, mask, (w-1, 0), BACKGROUND_ID, flags=4 | (255 << 8))
    cv2.floodFill(map_img, mask, (0, h-1), BACKGROUND_ID, flags=4 | (255 << 8))
    cv2.floodFill(map_img, mask, (w-1, h-1), BACKGROUND_ID, flags=4 | (255 << 8))

    fixed_details = None
    fixed_rgb = None
    if BEGINNER_MODE:
        map_img[source_ink] = LINE_ID
        # Make the hit map respect the visible boundary overlay. Otherwise a
        # one-pixel bridge can join two areas that look completely enclosed.
        visible_boundaries = cv2.morphologyEx(map_img, cv2.MORPH_GRADIENT,
            np.ones((3, 3), np.uint8)) > 0
        map_img[visible_boundaries] = LINE_ID
        # Close subpixel-looking contour gaps that disappear when downscaled
        # to a phone, so visually enclosed lobes are separate fill regions.
        closed_ink = cv2.morphologyEx(np.uint8(map_img == LINE_ID),
            cv2.MORPH_CLOSE, np.ones((3, 3), np.uint8))
        map_img[closed_ink > 0] = LINE_ID
        unsimplified = map_img.copy()
        source_centers = np.asarray(final_centers, dtype=np.uint8)
        map_img, final_centers = simplify_beginner_map(map_img, final_centers)
        fixed_details = (unsimplified < LINE_ID) & (map_img == LINE_ID)
        fixed_rgb = source_centers[unsimplified[fixed_details]]

    lines = np.zeros_like(map_img)
    gradient = cv2.morphologyEx(map_img, cv2.MORPH_GRADIENT, np.ones((3,3), np.uint8))
    lines[gradient > 0] = 255 
    if has_black_outline:
        lines[map_img == LINE_ID] = 255

    lines_rgba = np.zeros((h, w, 4), dtype=np.uint8)
    lines_rgba[lines > 0] = [0, 0, 0, 255] 
    if BEGINNER_MODE:
        # Preserve little eye highlights and decorative marks without asking a
        # child to find or tap them. The canvas already supports an RGBA overlay.
        lines_rgba[fixed_details, :3] = fixed_rgb
        lines_rgba[fixed_details, 3] = 255
        # Small enclosed spaces between branches/leaves are background. Render
        # them white so the tinted canvas does not suggest a missing target.
        lines_rgba[enclosed_background(map_img)] = [255, 255, 255, 255]

    number_data = []
    palette = [rgb_to_hex(c) for c in final_centers]
    unique_ids = np.unique(map_img)

    total_regions = 0
    total_labels = 0

    print("   - Grouping neighbors and placing labels...")
    for color_idx in unique_ids:
        if color_idx == BACKGROUND_ID or color_idx == LINE_ID:
            continue

        groups = group_neighboring_regions(map_img, color_idx, MIN_REGION_AREA,
            neighbor_gap=0 if BEGINNER_MODE else NEIGHBOR_GAP)

        for group in groups:
            total_regions += len(group['regions'])
            total_labels += 1

            cx, cy = group['best_label_pos']
            number_data.append({
                "number": int(color_idx + 1),
                "x": int(cx),
                "y": int(cy),
                "color_index": int(color_idx)
            })

    print(f"   - Consolidated {total_regions} regions into {total_labels} labels.")

    display_map = map_img.copy()
    if has_black_outline:
        display_map[display_map == LINE_ID] = darkest_idx
        
    clean_img_pixels = np.array([final_centers[l if l < len(final_centers) else 0] for l in display_map.flatten()])
    clean_img = clean_img_pixels.reshape(img.shape).astype(np.uint8)
    if BEGINNER_MODE:
        clean_img[map_img == LINE_ID] = [0, 0, 0]
        clean_img[fixed_details] = fixed_rgb
        clean_img[map_img == BACKGROUND_ID] = [255, 255, 255]
    
    cv2.imwrite(os.path.join(save_path, "original.png"), cv2.cvtColor(clean_img, cv2.COLOR_RGB2BGR))
    write_rgba_png(os.path.join(save_path, "lines.png"), lines_rgba)
    cv2.imwrite(os.path.join(save_path, "map.png"), map_img)
    
    debug_img = cv2.cvtColor(clean_img, cv2.COLOR_RGB2BGR)
    for item in number_data:
        cv2.putText(debug_img, str(item['number']), (item['x']-5, item['y']+3), 
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (255,255,255), 2)
        cv2.putText(debug_img, str(item['number']), (item['x']-5, item['y']+3), 
                    cv2.FONT_HERSHEY_SIMPLEX, 0.5, (0,0,0), 1)
        
    cv2.imwrite(os.path.join(save_path, "debug_preview.png"), debug_img)

    has_video = False
    if os.path.exists(vid_path):
        shutil.copy(vid_path, os.path.join(save_path, "reward.mp4"))
        has_video = True
    
    level_data = {
        "id": base_name,
        "palette": palette,
        "numbers": number_data,
        "dimensions": {"w": w, "h": h},
        "has_reward": has_video,
        "background_id": BACKGROUND_ID,
        "line_id": LINE_ID
    }
    
    with open(os.path.join(save_path, "data.json"), 'w') as f:
        json.dump(level_data, f, indent=2)

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description="Process paint-by-numbers images (IMPROVED)")
    parser.add_argument("--from", dest="start_from", type=int, default=1, help="Start from")
    parser.add_argument("--to", dest="end_at", type=int, default=100, help="End at")
    parser.add_argument("--beginner", action="store_true", help="Remove tiny numbered edge fragments and unused palette colors")
    args = parser.parse_args()
    BEGINNER_MODE = args.beginner

    if not os.path.exists(INPUT_DIR): os.makedirs(INPUT_DIR)
    if not os.path.exists(OUTPUT_DIR): os.makedirs(OUTPUT_DIR)

    def get_file_number(filename):
        match = re.match(r'^(\d+)', filename)
        return int(match.group(1)) if match else 0

    files = [f for f in os.listdir(INPUT_DIR) if f.lower().endswith(('.png', '.jpg', '.jpeg'))]
    files = [f for f in files if get_file_number(f) >= args.start_from and get_file_number(f) <= args.end_at]
    files.sort(key=get_file_number)

    print(f"Processing {len(files)} files (levels {args.start_from} to {args.end_at})...")
    print(f"Using IMPROVED settings: MERGE={MERGE_THRESHOLD}, GAP={NEIGHBOR_GAP}, DESPECKLE={DESPECKLE_SIZE}")
    for f in files: process_level(f)
